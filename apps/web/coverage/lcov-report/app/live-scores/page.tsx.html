            <CardContent>
              <div className="space-y-3 max-h-64 overflow-y-auto">

            </CardContent>
          </Card>
        </TabsContent>
      </Tabs>
 
      {/* Mid-Game Optimizer Modal */}
      <MidGameOptimizer
        isOpen={showOptimizer}
        onClose={() => setShowOptimizer(false)}
      />
    </div>
